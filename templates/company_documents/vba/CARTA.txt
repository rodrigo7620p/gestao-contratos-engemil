Option Explicit

' =====================================================
' GERAR CARTA
' =====================================================

Sub GerarCarta()

    Dim caminhoBase As String
    Dim caminhoControle As String
    Dim caminhoSaida As String

    Dim numeroAtual As Long
    Dim novoNumero As Long

    Dim anoAtual As String
    Dim numeroCarta As String
    Dim numeroArquivo As String

    Dim orgao As String
    Dim sigla As String
    Dim contrato As String
    Dim processo As String
    Dim centroCusto As String
    Dim objeto As String

    Dim etapa As Integer

    ' =====================================================
    ' ENGENHEIRO
    ' =====================================================

    Dim engenheiro As String
    Dim nacionalidade As String
    Dim titulo As String
    Dim cpf As String
    Dim crea As String

    ' =====================================================
    ' ASSINATURA
    ' =====================================================

    Dim responsavel As String
    Dim dados As String
    Dim cargo As String

    ' =====================================================
    ' CAMINHOS
    ' =====================================================

    caminhoBase = _
    "G:\Meu Drive\REDE\SISTEMA_DOC_ENGEMIL"

    caminhoControle = _
    caminhoBase & _
    "\01_CONTROLE\controle_carta.txt"

    caminhoSaida = _
    caminhoBase & _
    "\03_GERADOS\02_CARTAS\"

    ' =====================================================
    ' CRIA PASTAS
    ' =====================================================

    Call CriarPastaSeNaoExistir(caminhoBase)

    Call CriarPastaSeNaoExistir( _
    caminhoBase & "\01_CONTROLE")

    Call CriarPastaSeNaoExistir( _
    caminhoBase & "\03_GERADOS")

    Call CriarPastaSeNaoExistir( _
    caminhoBase & "\03_GERADOS\02_CARTAS")

    ' =====================================================
    ' DADOS COM BOTÃO VOLTAR
    ' =====================================================

    anoAtual = Year(Date)

    etapa = 1

    Do While etapa <= 5

        Select Case etapa

            Case 1

                orgao = PerguntarCampo( _
                "ÓRGÃO", _
                "Informe o órgão:", _
                orgao)

                If orgao = "<<" Then

                    MsgBox _
                    "Já está no primeiro campo.", _
                    vbExclamation

                Else

                    etapa = etapa + 1

                End If

            Case 2

                sigla = PerguntarCampo( _
                "SIGLA", _
                "Informe a sigla:", _
                sigla)

                If sigla = "<<" Then

                    etapa = etapa - 1

                Else

                    etapa = etapa + 1

                End If

            Case 3

                contrato = PerguntarCampo( _
                "CONTRATO", _
                "Informe o contrato:", _
                contrato)

                If contrato = "<<" Then

                    etapa = etapa - 1

                Else

                    etapa = etapa + 1

                End If

            Case 4

                processo = PerguntarCampo( _
                "PROCESSO", _
                "Informe o processo:", _
                processo)

                If processo = "<<" Then

                    etapa = etapa - 1

                Else

                    etapa = etapa + 1

                End If

            Case 5

                centroCusto = PerguntarCampo( _
                "CENTRO DE CUSTO", _
                "Informe o centro de custo:", _
                centroCusto)

                If centroCusto = "<<" Then

                    etapa = etapa - 1

                Else

                    etapa = etapa + 1

                End If

        End Select

    Loop

    ' =====================================================
    ' OBJETO LONGO
    ' =====================================================

    MsgBox _
    "O próximo bloco de notas será para o OBJETO.", _
    vbInformation

    objeto = LerTextoArquivo( _
    "objeto_carta.txt", _
    "COLE OU DIGITE O OBJETO AQUI.")

    ' =====================================================
    ' ENGENHEIRO
    ' =====================================================

    Call SelecionarEngenheiro( _
    engenheiro, _
    nacionalidade, _
    titulo, _
    cpf, _
    crea)

    ' =====================================================
    ' ASSINATURA
    ' =====================================================

    Call SelecionarAssinatura( _
    responsavel, _
    dados, _
    cargo)

    If centroCusto = "" Then

        MsgBox _
        "Operação cancelada.", _
        vbExclamation

        Exit Sub

    End If

    ' =====================================================
    ' NUMERAÇÃO
    ' =====================================================

    numeroAtual = LerNumeroControle( _
    caminhoControle)

    novoNumero = numeroAtual + 1

    numeroCarta = _
    "CA-" & _
    anoAtual & "-" & _
    Format(novoNumero, "0000") & _
    "/ENGEMIL/DCONT/" & _
    contrato & "/" & _
    sigla

    ' =====================================================
    ' NOME DO ARQUIVO
    ' =====================================================

    numeroArquivo = _
    Replace(numeroCarta, "/", "-")

    ' =====================================================
    ' SUBSTITUIÇÕES
    ' =====================================================

    Call SubstituirTextoCompleto( _
    "{{NUMERO_CARTA}}", _
    numeroCarta)

    Call SubstituirTextoCompleto( _
    "{{DATA}}", _
    Format(Date, "dd/mm/yyyy"))

    Call SubstituirTextoCompleto( _
    "{{DATA_EXTENSO}}", _
    DataPorExtenso())

    Call SubstituirTextoCompleto( _
    "{{ORGAO}}", _
    orgao)

    Call SubstituirTextoCompleto( _
    "{{SIGLA}}", _
    sigla)

    Call SubstituirTextoCompleto( _
    "{{CONTRATO}}", _
    contrato)

    Call SubstituirTextoCompleto( _
    "{{PROCESSO}}", _
    processo)

    Call SubstituirTextoCompleto( _
    "{{CENTRO_CUSTO}}", _
    centroCusto)

    Call SubstituirTextoCompleto( _
    "{{OBJETO}}", _
    objeto)

    ' =====================================================
    ' ENGENHEIRO
    ' =====================================================

    Call SubstituirTextoCompleto( _
    "{{ENGENHEIRO}}", _
    engenheiro)

    Call SubstituirTextoCompleto( _
    "{{NACIONALIDADE}}", _
    nacionalidade)

    Call SubstituirTextoCompleto( _
    "{{TITULO}}", _
    titulo)

    Call SubstituirTextoCompleto( _
    "{{CPF}}", _
    cpf)

    Call SubstituirTextoCompleto( _
    "{{CREA}}", _
    crea)

    ' =====================================================
    ' ASSINATURA
    ' =====================================================

    Call SubstituirTextoCompleto( _
    "{{RESPONSAVEL}}", _
    responsavel)

    Call SubstituirTextoCompleto( _
    "{{DADOS}}", _
    dados)

    Call SubstituirTextoCompleto( _
    "{{CARGO}}", _
    cargo)

    ' =====================================================
    ' SALVAR WORD
    ' =====================================================

    ActiveDocument.SaveAs2 _
    FileName:= _
    caminhoSaida & numeroArquivo & ".docx", _
    FileFormat:=wdFormatXMLDocument

    ' =====================================================
    ' GERAR PDF
    ' =====================================================

    ActiveDocument.ExportAsFixedFormat _
    OutputFileName:= _
    caminhoSaida & numeroArquivo & ".pdf", _
    ExportFormat:=wdExportFormatPDF

    ' =====================================================
    ' SALVAR CONTROLE
    ' =====================================================

    Call SalvarNumeroControle( _
    caminhoControle, _
    novoNumero)

    MsgBox _
    "CARTA GERADA COM SUCESSO!" & _
    vbCrLf & vbCrLf & _
    numeroCarta, _
    vbInformation

End Sub

' =====================================================
' INPUTBOX COM VOLTAR
' =====================================================

Function PerguntarCampo( _
titulo As String, _
mensagem As String, _
valorAnterior As String) As String

    Dim resposta As String

    resposta = InputBox( _
    mensagem & vbCrLf & vbCrLf & _
    "Digite << para VOLTAR.", _
    titulo, _
    valorAnterior)

    PerguntarCampo = resposta

End Function

' =====================================================
' LEITOR UTF-8
' =====================================================

Function LerTextoArquivo( _
nomeArquivo As String, _
textoInicial As String) As String

    Dim caminhoTemp As String
    Dim shell As Object
    Dim stream As Object

    caminhoTemp = _
    Environ("TEMP") & "\" & nomeArquivo

    Set stream = _
    CreateObject("ADODB.Stream")

    stream.Type = 2
    stream.Charset = "utf-8"

    stream.Open
    stream.WriteText textoInicial
    stream.SaveToFile caminhoTemp, 2
    stream.Close

    Set shell = _
    CreateObject("WScript.Shell")

    shell.Run _
    "notepad.exe " & _
    Chr(34) & caminhoTemp & Chr(34), _
    1, _
    True

    Set stream = _
    CreateObject("ADODB.Stream")

    stream.Type = 2
    stream.Charset = "utf-8"

    stream.Open
    stream.LoadFromFile caminhoTemp

    LerTextoArquivo = stream.ReadText

    stream.Close

End Function

' =====================================================
' DATA POR EXTENSO
' =====================================================

Function DataPorExtenso() As String

    Dim meses As Variant

    meses = Array( _
    "janeiro", "fevereiro", "março", _
    "abril", "maio", "junho", _
    "julho", "agosto", "setembro", _
    "outubro", "novembro", "dezembro")

    DataPorExtenso = _
    "Brasília/DF., " & _
    Day(Date) & " de " & _
    meses(Month(Date) - 1) & _
    " de " & _
    Year(Date) & "."

End Function

' =====================================================
' LER CONTROLE
' =====================================================

Function LerNumeroControle( _
caminhoArquivo As String) As Long

    Dim arquivo As Integer
    Dim conteudo As String

    If Dir(caminhoArquivo) = "" Then

        arquivo = FreeFile

        Open caminhoArquivo _
        For Output As #arquivo

        Print #arquivo, 0

        Close #arquivo

    End If

    arquivo = FreeFile

    Open caminhoArquivo _
    For Input As #arquivo

    Line Input #arquivo, conteudo

    Close #arquivo

    If Trim(conteudo) = "" Then

        LerNumeroControle = 0

    Else

        LerNumeroControle = CLng(conteudo)

    End If

End Function

' =====================================================
' SALVAR CONTROLE
' =====================================================

Sub SalvarNumeroControle( _
caminhoArquivo As String, _
numero As Long)

    Dim arquivo As Integer

    arquivo = FreeFile

    Open caminhoArquivo _
    For Output As #arquivo

    Print #arquivo, numero

    Close #arquivo

End Sub

' =====================================================
' SUBSTITUIR TEXTO COMPLETO
' =====================================================

Sub SubstituirTextoCompleto( _
localizar As String, _
substituirPor As String)

    Dim rngStory As Range
    Dim rngAtual As Range

    Dim secao As Section
    Dim cabecalho As HeaderFooter
    Dim rodape As HeaderFooter

    Dim shp As Shape

    For Each rngStory In _
    ActiveDocument.StoryRanges

        Set rngAtual = rngStory

        Do While Not rngAtual Is Nothing

            Call SubstituirEmRangeSeguro( _
            rngAtual, _
            localizar, _
            substituirPor)

            Set rngAtual = _
            rngAtual.NextStoryRange

        Loop

    Next rngStory

    For Each secao In _
    ActiveDocument.Sections

        For Each cabecalho In _
        secao.Headers

            Call SubstituirEmRangeSeguro( _
            cabecalho.Range, _
            localizar, _
            substituirPor)

            For Each shp In cabecalho.Shapes

                If shp.TextFrame.HasText Then

                    Call SubstituirEmRangeSeguro( _
                    shp.TextFrame.TextRange, _
                    localizar, _
                    substituirPor)

                End If

            Next shp

        Next cabecalho

        For Each rodape In _
        secao.Footers

            Call SubstituirEmRangeSeguro( _
            rodape.Range, _
            localizar, _
            substituirPor)

            For Each shp In rodape.Shapes

                If shp.TextFrame.HasText Then

                    Call SubstituirEmRangeSeguro( _
                    shp.TextFrame.TextRange, _
                    localizar, _
                    substituirPor)

                End If

            Next shp

        Next rodape

    Next secao

End Sub

' =====================================================
' SUBSTITUIR RANGE
' =====================================================

Sub SubstituirEmRangeSeguro( _
rngOrigem As Range, _
localizar As String, _
substituirPor As String)

    Dim rngBusca As Range

    Set rngBusca = rngOrigem.Duplicate

    With rngBusca.Find

        .ClearFormatting
        .Text = localizar
        .Forward = True
        .Wrap = wdFindStop
        .Format = False
        .MatchCase = False
        .MatchWholeWord = False

    End With

    Do While rngBusca.Find.Execute

        rngBusca.Text = substituirPor

        rngBusca.Collapse wdCollapseEnd

    Loop

End Sub

' =====================================================
' CRIAR PASTA
' =====================================================

Sub CriarPastaSeNaoExistir( _
caminho As String)

    Dim fso As Object

    Set fso = _
    CreateObject( _
    "Scripting.FileSystemObject")

    If Not _
    fso.FolderExists(caminho) Then

        fso.CreateFolder caminho

    End If

End Sub
